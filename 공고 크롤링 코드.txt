(async function crawlDetailedJobs() {
  console.log("=== 링커리어 공고 상세 일정 정밀 수집 시작 ===");

  // 1. 현재 화면의 공고 카드 링크 수집 (중복 제거)
  const cardLinks = Array.from(document.querySelectorAll('a[href*="/activity/"]'));
  const uniqueJobs = new Map();

  cardLinks.forEach(el => {
    const rawUrl = el.href.split('?')[0];
    const textLines = el.innerText.trim().split('\n').map(t => t.trim()).filter(Boolean);
    if (rawUrl && !uniqueJobs.has(rawUrl) && textLines.length > 0) {
      uniqueJobs.set(rawUrl, {
        rawCompany: textLines[0],
        rawTitle: textLines.slice(1).join(" "),
        link: rawUrl
      });
    }
  });

  const jobList = Array.from(uniqueJobs.values());
  console.log(`총 ${jobList.length}개의 고유 공고를 확인했습니다. 상세 페이지 탐색 시작...`);

  const results = [];

  // 2. 각 상세 페이지 순회 및 정밀 데이터 파싱
  for (let i = 0; i < jobList.length; i++) {
    const item = jobList[i];
    let startDate = "";
    let endDate = "";
    let companyName = item.rawCompany;
    let titleName = item.rawTitle;

    try {
      const res = await fetch(item.link);
      const html = await res.text();
      const parser = new DOMParser();
      const doc = parser.parseFromString(html, 'text/html');

      // A. Next.js 내장 데이터(__NEXT_DATA__)에서 직접 JSON 추출 (가장 정확함)
      const nextDataEl = doc.getElementById('__NEXT_DATA__');
      if (nextDataEl) {
        try {
          const nextJson = JSON.parse(nextDataEl.textContent);
          // pageProps 내의 recruit/activity 객체 탐색
          const props = nextJson.props?.pageProps;
          const activity = props?.activity || props?.data || props?.recruit;
          
          if (activity) {
            if (activity.organizationName) companyName = activity.organizationName;
            if (activity.title) titleName = activity.title;
            if (activity.startDate) startDate = activity.startDate.substring(0, 10);
            if (activity.endDate) endDate = activity.endDate.substring(0, 10);
          }
        } catch (e) {
          // JSON 파싱 에러 시 HTML 텍스트 정규식으로 전환
        }
      }

      // B. JSON 데이터 추출이 안 되었을 경우: 이미지 속 화면의 "시작일 YYYY.MM.DD", "마감일 YYYY.MM.DD" 직접 텍스트 매칭
      const bodyText = doc.body.innerText;
      
      if (!startDate) {
        const startMatch = bodyText.match(/시작일\s*[:|]?\s*(\d{4}[.\-/]\d{2}[.\-/]\d{2})/);
        if (startMatch) startDate = startMatch[1].replace(/[.\/]/g, '-');
      }

      if (!endDate) {
        const endMatch = bodyText.match(/마감일\s*[:|]?\s*(\d{4}[.\-/]\d{2}[.\-/]\d{2})/);
        if (endMatch) endDate = endMatch[1].replace(/[.\/]/g, '-');
      }

      // 만약 '시작일/마감일' 텍스트가 없고 범위 형태(2026.09.01 ~ 2026.09.16)로만 적힌 경우 대비
      if (!startDate || !endDate) {
        const rangeMatch = bodyText.match(/(\d{4}[.\-/]\d{2}[.\-/]\d{2})\s*[~-]\s*(\d{4}[.\-/]\d{2}[.\-/]\d{2})/);
        if (rangeMatch) {
          if (!startDate) startDate = rangeMatch[1].replace(/[.\/]/g, '-');
          if (!endDate) endDate = rangeMatch[2].replace(/[.\/]/g, '-');
        }
      }

      console.log(`[${i + 1}/${jobList.length}] ${companyName} | 시작: ${startDate || '미정'} | 마감: ${endDate || '미정'}`);

    } catch (err) {
      console.warn(`[${i + 1}/${jobList.length}] ${item.link} 로드 실패:`, err);
    }

    results.push({
      company: companyName.replace(/[\[\]]/g, '').trim(),
      title: titleName.trim() || item.rawTitle,
      startDate: startDate,
      endDate: endDate,
      link: item.link
    });

    // 서버 요청 제한 방지를 위한 짧은 딜레이 (0.2초)
    await new Promise(r => setTimeout(r, 200));
  }

  console.log("=== 수집 완료 ===");
  console.table(results);

  // 3. 파일 다운로드
  const blob = new Blob([JSON.stringify(results, null, 2)], { type: 'application/json' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'jobs_data.json';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  console.log("다운로드 폴더에 'jobs_data.json'이 생성되었습니다.");
})();
